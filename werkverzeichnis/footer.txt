<script>
  // Aktuelles Jahr beim Copyright
  document.getElementById('year').textContent = new Date().getFullYear();

  // Up Top Button
  var upTopBtn = document.getElementById("UpTop");

  // Wenn der Benutzer 100px vom oberen Rand des Dokuments nach unten scrollt, zeigen Sie die Schaltfläche
  window.onscroll = function() {
    if (document.body.scrollTop > 100 || document.documentElement.scrollTop > 100) {
      upTopBtn.style.display = "block";
    } else {
      upTopBtn.style.display = "none";
    }
  };

  // Wenn der Benutzer auf die Schaltfläche klickt, blättert es an den Anfang des Dokuments
  upTopBtn.onclick = function() {
    window.scrollTo({
      top: 0,
      behavior: "smooth" // Für einen reibungslosen Bildlauf
    });
  };
</script>
